datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

generator client {
  provider = "prisma-client-js"
}

// --------------------------------------------------------
// USER & REPUTATION
// --------------------------------------------------------
model User {
  id              String         @id @default(uuid())
  email           String         @unique
  username        String         @unique
  tacticalIqScore Int            @default(50) // 0-100 rolling score based on prediction accuracy
  badge           RefereeBadge   @default(SPECTATOR)
  debates         Debate[]
  votes           DebateVote[]
  claimVotes      ClaimVote[]
  createdAt       DateTime       @default(now())
  updatedAt       DateTime       @updatedAt
}

enum RefereeBadge {
  SPECTATOR
  ASSISTANT_REF
  CHIEF_VAR
}

// --------------------------------------------------------
// FOOTBALL METRICS & TELEMETRY
// --------------------------------------------------------
model Player {
  id           String        @id @default(uuid())
  name         String        @unique
  currentClub  String
  primaryRole  String
  matchEvents  MatchEvent[]
  debates      Debate[]
  createdAt    DateTime      @default(now())
}

model MatchEvent {
  id           String    @id @default(uuid())
  playerId     String
  player       Player    @relation(fields: [playerId], references: [id], onDelete: Cascade)
  matchId      String
  season       String    // e.g., "2024-2025"
  eventType    EventType // PASS, SHOT, CARRY, RECOVERY
  minute       Int
  xCoord       Float     // 0.0 - 100.0 (Pitch coordinate standard)
  yCoord       Float     // 0.0 - 100.0
  endXCoord    Float?    // Applicable for passes/carries
  endYCoord    Float?
  isProgressive Boolean  @default(false)
  xG           Float?
  xA           Float?
  zone         String    // "box_central", "half_space_left", "deep_third"

  @@index([playerId, season])
  @@index([playerId, zone])
  @@index([xCoord, yCoord])
}

enum EventType {
  PASS
  SHOT
  CARRY
  RECOVERY
  BOX_TOUCH
}

// --------------------------------------------------------
// DEBATES, VERDICTS & VOTING
// --------------------------------------------------------
model Debate {
  id               String          @id @default(uuid())
  authorId         String
  author           User            @relation(fields: [authorId], references: [id], onDelete: Cascade)
  title            String
  thesis           String
  focusPlayerId    String?
  focusPlayer      Player?         @relation(fields: [focusPlayerId], references: [id])
  
  // VAR Agent Verdict Results
  varStatus        VarStatus       @default(PENDING)
  credibilityScore Int?            // 0 - 100
  varSummary       String?
  hotTakeIndex     Float           @default(0.0) // Abs divergence: |VAR Score - Consensus|
  
  claims           Claim[]
  votes            DebateVote[]
  
  createdAt        DateTime        @default(now())
  updatedAt        DateTime        @updatedAt

  @@index([varStatus])
}

enum VarStatus {
  PENDING
  UNDER_REVIEW
  CONFIRMED
  OVERTURNED
  INCONCLUSIVE
}

model Claim {
  id               String          @id @default(uuid())
  debateId         String
  debate           Debate          @relation(fields: [debateId], references: [id], onDelete: Cascade)
  claimText        String
  metricEvaluated  String          // "box_touches", "progressive_carries"
  observedValue    Float?
  benchmarkValue   Float?
  claimVerdict     ClaimVerdict    @default(PENDING)
  reasoning        String?
  votes            ClaimVote[]

  createdAt        DateTime        @default(now())
}

enum ClaimVerdict {
  PENDING
  VERIFIED
  PARTIALLY_TRUE
  REFUTED
}

// --------------------------------------------------------
// COMMUNITY JURY VOTING
// --------------------------------------------------------
model DebateVote {
  id        String       @id @default(uuid())
  userId    String
  user      User         @relation(fields: [userId], references: [id], onDelete: Cascade)
  debateId  String
  debate    Debate       @relation(fields: [debateId], references: [id], onDelete: Cascade)
  voteType  VoteType     // UPHOLD (Agree with user thesis) or OVERTURN (Disagree)
  createdAt DateTime     @default(now())

  @@unique([userId, debateId])
}

model ClaimVote {
  id        String       @id @default(uuid())
  userId    String
  user      User         @relation(fields: [userId], references: [id], onDelete: Cascade)
  claimId   String
  claim     Claim        @relation(fields: [claimId], references: [id], onDelete: Cascade)
  isAccurate Boolean
  createdAt DateTime     @default(now())

  @@unique([userId, claimId])
}

enum VoteType {
  UPHOLD
  OVERTURN
}
